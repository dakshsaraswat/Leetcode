class Solution:
    def moveZeroes(self, nums: list[int]) -> None:
        """
        Do not return anything, modify nums in-place instead.
        """
        # Pointer to track the position of the next non-zero element
        non_zero_pos = 0
        
        # Iterate through the array with the current pointer
        for i in range(len(nums)):
            if nums[i] != 0:
                # Swap the non-zero element with the element at non_zero_pos
                nums[non_zero_pos], nums[i] = nums[i], nums[non_zero_pos]
                # Move the non-zero tracker forward
                non_zero_pos += 1
