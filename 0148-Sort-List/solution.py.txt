# Definition for singly-linked list.
# class ListNode:
#     def __init__(self, val=0, next=None):
#         self.val = val
#         self.next = next

class Solution:
    def sortList(self, head: Optional[ListNode]) -> Optional[ListNode]:
        # Fast exit for empty list or single node
        if not head or not head.next:
            return head
        
        # Step 1: Collect all node objects into an array
        nodes = []
        curr = head
        while curr:
            nodes.append(curr)
            curr = curr.next
            
        # Step 2: Sort the node objects using Python's highly optimized Timsort
        nodes.sort(key=lambda x: x.val)
        
        # Step 3: Re-link the nodes sequentially in sorted order
        for i in range(len(nodes) - 1):
            nodes[i].next = nodes[i + 1]
            
        # Step 4: Terminate the tail of the sorted list
        nodes[-1].next = None
        
        return nodes[0]
